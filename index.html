// 自動向 Google 查詢可用模型，並排除免費額度為 0 的特殊模型 async function fetchAvailableModels() { const apiKey = document.getElementById('apiKey').value.trim(); const statusEl = document.getElementById('modelStatus'); const selectEl = document.getElementById('model'); if (!apiKey) { statusEl.textContent = '⚠️ 請先輸入 API Key'; return; } statusEl.textContent = '⏳ 正在向 Google 查詢可用模型...'; try { const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models?key=${apiKey}`); const data = await res.json(); if (data.error) throw new Error(data.error.message); // 過濾出標準 gemini 模型，排除 omni、tts、image、thinking 等免費額度為 0 或不支援工具的模型 const validModels = (data.models || []) .filter(m => m.supportedGenerationMethods && m.supportedGenerationMethods.includes('generateContent')) .map(m => m.name.replace('models/', '')) .filter(name => name.startsWith('gemini') && !name.includes('omni') && !name.includes('vision') && !name.includes('embedding') && !name.includes('tts') && !name.includes('image') ); // 排序：優先選擇 gemini-3-flash / gemini-flash-latest / lite 等有免費額度的模型 validModels.sort((a, b) => { const score = (n) => { let s = 0; if (n === 'gemini-flash-latest' || n === 'gemini-3-flash-preview') s += 50; if (n.includes('3') && n.includes('flash')) s += 30; if (n.includes('lite')) s += 20; if (n.includes('flash')) s += 10; if (n.includes('pro') || n.includes('exp')) s -= 15; // pro 與 exp 常有免費額度限制，往後排 return s; }; return score(b) - score(a); }); selectEl.innerHTML = ''; validModels.forEach(modelName => { const opt = document.createElement('option'); opt.value = modelName; opt.textContent = modelName; selectEl.appendChild(opt); }); // 如果之前手動選過不會報錯的模型就保留，否則選排序後的第一個 const savedModel = localStorage.getItem('gemini_agent_model'); if (savedModel && validModels.includes(savedModel) && !savedModel.includes('omni')) { selectEl.value = savedModel; } else if (validModels.length > 0) { selectEl.value = validModels[0]; localStorage.setItem('gemini_agent_model', validModels[0]); } statusEl.style.color = '#16a34a'; statusEl.textContent = `✅ 已載入 ${validModels.length} 個模型（目前：${selectEl.value}）`; } catch (err) { statusEl.style.color = '#dc2626'; statusEl.textContent = `❌ 模型清單讀取失敗: ${err.message}`; } }
